                        <span id="status-${qrId}" style="color: green; font-size: 12px; display: none;">✓ Tallennettu</span>

                    </div>

                `;

                

                list.appendChild(config);

            }

        }


        function clearURL(qrId) {

            if (confirm(`Haluatko varmasti tyhjentää QR ${qrId} URL:n?`)) {

                saveRedirectURL(qrId, '');

                generateConfigList();

            }

        }


        function downloadQR(qrId) {

            const canvas = document.getElementById(`qr-${qrId}`);

            const url = canvas.toDataURL('image/png');

            const link = document.createElement('a');

            link.download = `QR-${qrId}.png`;

            link.href = url;

            link.click();

        }


        function downloadAllQRs() {

            if (!confirm('Haluatko ladata kaikki 100 QR-koodia? Tämä voi kestää hetken.')) {

                return;

            }

            

            let downloaded = 0;

            const interval = setInterval(() => {

                if (downloaded >= TOTAL_QR_CODES) {

                    clearInterval(interval);

                    alert('Kaikki QR-koodit ladattu!');

                    return;

                }

                

                downloaded++;

                const qrId = String(downloaded).padStart(3, '0');

                downloadQR(qrId);

            }, 200); // 200ms viive jokaisen latauksen välillä

        }


        function filterConfigs() {

            const search = document.getElementById('search').value.toLowerCase();

            const configs = document.querySelectorAll('.url-config');

            

            configs.forEach(config => {

                const qrId = config.dataset.qrId.toLowerCase();

                config.style.display = qrId.includes(search) ? 'block' : 'none';

            });

        }


        function exportConfig() {

            const urls = getRedirectURLs();

            const dataStr = JSON.stringify(urls, null, 2);

            const dataBlob = new Blob([dataStr], {type: 'application/json'});

            const url = URL.createObjectURL(dataBlob);

            const link = document.createElement('a');

            link.href = url;

            link.download = `qr-config-${new Date().toISOString().split('T')[0]}.json`;

            link.click();

        }


        function importConfig(event) {

            const file = event.target.files[0];

            if (!file) return;


            const reader = new FileReader();

            reader.onload = function(e) {

                try {

                    const config = JSON.parse(e.target.result);

                    localStorage.setItem('qr-redirects', JSON.stringify(config));

                    alert('Asetukset tuotu onnistuneesti!');

                    generateConfigList();

                    updateStats();

                } catch (error) {

                    alert('Virhe asetuksia tuotaessa: ' + error.message);

                }

            };

            reader.readAsText(file);

        }


        function updateStats() {

            const urls = getRedirectURLs();

            const configured = Object.keys(urls).length;

            const unconfigured = TOTAL_QR_CODES - configured;

            

            document.getElementById('configured-qr').textContent = configured;

            document.getElementById('unconfigured-qr').textContent = unconfigured;

        }


        function handleRedirect() {

            const urlParams = new URLSearchParams(window.location.search);

            let qrId = urlParams.get('qr');

            

            if (qrId) {

                // Normalisoi QR ID kolminumeroiseksi (esim. "35" -> "035")

                qrId = String(qrId).padStart(3, '0');

                console.log('QR redirect: qrId =', qrId);

                

                const urls = getRedirectURLs();

                console.log('Kaikki URL:t:', urls);

                const targetURL = urls[qrId];

                console.log('Haettu URL avaimella', qrId, ':', targetURL);

                

                if (targetURL) {

                    console.log('Ohjataan:', targetURL);

                    window.location.href = targetURL;

                } else {

                    console.log('URL ei löytynyt!');

                    document.body.innerHTML = `

                        <div style="display: flex; justify-content: center; align-items: center; height: 100vh; text-align: center; font-family: sans-serif;">

                            <div>

                                <h2 style="color: #dc3545;">⚠️ QR-koodi ${qrId} ei ole konfiguroitu</h2>

                                <p>Ota yhteyttä järjestelmän ylläpitäjään.</p>

                                <p style="font-size: 12px; color: #666; margin-top: 20px;">Debug: Avain "${qrId}" ei löydy localStoragesta</p>

                            </div>

                        </div>

                    `;

                }

            }

        }
